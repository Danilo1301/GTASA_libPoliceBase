static std::vector<std::string> splitStringBySpaces(const std::string& str)
{
    std::vector<std::string> result;
    std::istringstream iss(str);
    std::string word;

    // Usa o istringstream para separar a string em palavras
    while (iss >> word) {
        result.push_back(word);
    }

    return result;
}
